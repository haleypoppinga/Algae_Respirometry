Clean the Data

HoboData<- function(file) { # create a funciton to edit all the csv files so they are the same
  df<-read.csv(file) # read csv files
  columns<-c(grep("^Date", names(df), value = TRUE)[1], # find only 3 columns needed
             grep("^Temperature", names(df), value = TRUE)[1],
             grep("^Light", names(df), value = TRUE)[1])
  if (anyNA(columns)) { # check for NAs
    stop("Missing column in: ", file)
  }
  
  df<-df |> # clean the data 
    select(all_of(columns)) |> # only keep date temp lux columns for all files
    set_names(c("Date_Time", "Temperature_C", "Light_Lux")) |> # rename them b/c names were bad
    mutate(Date_Time = parse_date_time(Date_Time, # convert date and time using lubridate
                                       orders = c("mdY HMS", "mdY HM", "Ymd HMS", "Ymd HM"), # convert character strings into POSIXct date time values, formats to recognize
                                       tz = "Pacific/Honolulu"), # HI time zone (HST)
           Temperature_C = as.numeric(Temperature_C), # convert to numeric values
           Light_Lux = as.numeric(Light_Lux),
           source_file = basename(file)) |>  # create new column to id which csv corresponds to which row
    filter(Date_Time >= min(Date_Time, na.rm = TRUE) + minutes(30),
           Date_Time <= max(Date_Time, na.rm = TRUE) - hour(1)) # remove first 30 min and last hourfrom each file because automatic logging may have started during deployment
  return(df)
}

Read and Clean PAR Data

# PAR data to cross-calibrate
PAR_path<-here("Data","Hobo_Files", "PAR2026")

PARData <- function(file) { # function to clean PAR data
  df <- read.csv(file)
  df <- df |>
    select(Date_Time = starts_with("Date.Time"),
           Underwater_PAR = starts_with("Underwater.PAR"),
           Air_PAR = starts_with("In.Air.PAR"), # offshore PAR data doesnt have air par column so needed to do this
           Temperature_C = starts_with("Temperature"),
           Tilt_deg = starts_with("Tilt")) |> 
    #set_names(c("Date_Time","Underwater_PAR","Air_PAR","Temperature_C","Tilt_deg",
                #"DLI_Underwater","Accumulated_PAR")) |> # rename columns
    mutate(Date_Time = mdy_hms(Date_Time, tz = "Pacific/Honolulu"), # convert time so its the same as other files
      across(-Date_Time, ~ parse_number(as.character(.x))))
  
  if (!"Air_PAR" %in% names(df)) { # adds Air_PAR column b/c offshore file is missing it, this helps for the calibration code later
    df$Air_PAR <- NA_real_
  }
  
  return(df)
}

#plot PAR μmol/m²/s by light lux
#fit an exponential function following Long et al. 2012
mod_off <- nls(Underwater_PAR ~ A1*exp(-Light_Lux/t1) +y0,  # PAR in μmol/m²/s
           data = Light_off, start = list(A1 = -84000, t1 = 84000, y0 = 84000)) #t1 = min&max difference, A1=min-max, y0=maxPAR


# plot lux and PAR
ggplot(Light_off, aes(x = Light_Lux, y = Underwater_PAR)) + # Inshore
  geom_point() +
  #geom_line(aes(y = predict(mod_off)),color = "red", linewidth = 1) +
  geom_function(fun = function(x) 
    {coef(mod_off)["A1"] *exp(-x / coef(mod_off)["t1"]) + coef(mod_off)["y0"]},
                color = "red", linewidth = 1) +
  labs(x = "Light (lux)",y = "Underwater PAR (µmol/m²/s)")
# calibrate the light intensities to PAR based on this fit
m_off<-summary(mod_off)$coefficients[,1] # pull out the coefficients from the model

Notes:
* PAR = A1 e^{-Lux/t1}+y0
* A1: Controls the magnitude and direction of the exponential component
* t1: Controls how quickly the curve approaches its asymptote as lux increases
* y0: The horizontal asymptote that PAR approaches at high lux, assuming t1 is positive (i think these are true)
